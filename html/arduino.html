🔌 Arduino

Conceito, características, funcionamento e exemplos práticos de programação do Arduino aplicados à leitura de sensores em projetos de IoT e Automação Industrial.

O que é o Arduino?

Arduino é uma plataforma de prototipagem eletrônica de hardware livre (open-source), composta por uma placa com um microcontrolador e um ambiente de desenvolvimento integrado (IDE) próprio, baseado em uma linguagem simplificada derivada de C/C++.

Criado para facilitar o acesso à eletrônica embarcada, o Arduino permite ler sinais de sensores (entradas), processar essas informações e controlar atuadores como motores, relés e LEDs (saídas), sendo amplamente utilizado em projetos educacionais, de automação residencial, robótica e Internet das Coisas (IoT).

Como Funciona

O código-fonte, chamado de sketch, é escrito na IDE do Arduino e enviado para a placa através de uma porta USB. O programa é dividido basicamente em duas funções: setup(), executada uma única vez para inicializar pinos e bibliotecas, e loop(), executada repetidamente enquanto a placa estiver ligada, lendo sensores e controlando saídas em tempo real.

Características Técnicas (Arduino Uno)

  • Microcontrolador ATmega328P
  • Tensão de operação 5V
  • Tensão de entrada recomendada 7V a 12V
  • Pinos digitais I/O 14 (6 com PWM)
  • Pinos de entrada analógica 6 (A0 a A5)
  • Memória Flash 32 KB
  • Memória SRAM 2 KB
  • Clock 16 MHz
  • Comunicação USB, UART, I2C, SPI

DHT11 - Temperatura e Umidade

Lê a temperatura e a umidade do ar usando a biblioteca DHT.

Ligações: Pino de dados do DHT11 → Pino digital 2 do Arduino (com resistor de pull-up de 10kΩ, se necessário)
// Exemplo: Leitura de Temperatura e Umidade com o sensor DHT11
#include <DHT.h>

#define PINO_DHT 2      // Pino digital conectado ao sensor
#define TIPO_DHT DHT11  // Define o modelo do sensor

DHT dht(PINO_DHT, TIPO_DHT);

void setup() {
  Serial.begin(9600);
  dht.begin(); // Inicializa o sensor
}

void loop() {
  float umidade = dht.readHumidity();        // Lê a umidade relativa (%)
  float temperatura = dht.readTemperature(); // Lê a temperatura (°C)

  // Verifica se a leitura falhou
  if (isnan(umidade) || isnan(temperatura)) {
    Serial.println("Falha ao ler o sensor DHT11!");
    return;
  }

  Serial.print("Temperatura: ");
  Serial.print(temperatura);
  Serial.print(" C | Umidade: ");
  Serial.print(umidade);
  Serial.println(" %");

  delay(2000); // Aguarda 2 segundos entre leituras
}

LM35 - Temperatura Analógica

Converte a tensão de saída do LM35 em graus Celsius.

Ligações: Saída do LM35 → Pino analógico A0 do Arduino
// Exemplo: Leitura de Temperatura Analógica com o sensor LM35
const int PINO_LM35 = A0; // Pino analógico conectado à saída do sensor

void setup() {
  Serial.begin(9600);
}

void loop() {
  int leituraADC = analogRead(PINO_LM35);      // Lê o valor bruto (0-1023)
  float tensao = leituraADC * (5.0 / 1023.0);  // Converte para tensão (0-5V)
  float temperatura = tensao * 100.0;          // LM35: 10 mV por °C

  Serial.print("Temperatura: ");
  Serial.print(temperatura);
  Serial.println(" C");

  delay(1000);
}

HC-SR04 - Medição de Distância

Calcula a distância a partir do tempo de retorno do pulso ultrassônico.

Ligações: Trigger → Pino digital 9 | Echo → Pino digital 10
// Exemplo: Medição de Distância com o sensor ultrassônico HC-SR04
const int PINO_TRIGGER = 9;
const int PINO_ECHO = 10;

void setup() {
  Serial.begin(9600);
  pinMode(PINO_TRIGGER, OUTPUT);
  pinMode(PINO_ECHO, INPUT);
}

void loop() {
  // Envia um pulso ultrassônico de 10 microssegundos
  digitalWrite(PINO_TRIGGER, LOW);
  delayMicroseconds(2);
  digitalWrite(PINO_TRIGGER, HIGH);
  delayMicroseconds(10);
  digitalWrite(PINO_TRIGGER, LOW);

  // Mede o tempo de retorno do eco (em microssegundos)
  long duracao = pulseIn(PINO_ECHO, HIGH);

  // Velocidade do som ~0,034 cm/us, dividido por 2 (ida e volta)
  float distancia = duracao * 0.034 / 2;

  Serial.print("Distancia: ");
  Serial.print(distancia);
  Serial.println(" cm");

  delay(500);
}

PIR HC-SR501 - Detecção de Presença

Aciona um LED sempre que movimento é detectado no ambiente.

Ligações: Saída OUT do PIR → Pino digital 3 | LED → Pino digital 13
// Exemplo: Detecção de Presença com o sensor PIR HC-SR501
const int PINO_PIR = 3;
const int PINO_LED = 13;

void setup() {
  Serial.begin(9600);
  pinMode(PINO_PIR, INPUT);
  pinMode(PINO_LED, OUTPUT);
}

void loop() {
  int movimento = digitalRead(PINO_PIR); // HIGH = movimento detectado

  if (movimento == HIGH) {
    digitalWrite(PINO_LED, HIGH);
    Serial.println("Movimento detectado!");
  } else {
    digitalWrite(PINO_LED, LOW);
  }

  delay(200);
}

LDR - Controle Automático de Iluminação

Aciona um relé (lâmpada) automaticamente quando o ambiente escurece.

Ligações: LDR em divisor de tensão → Pino analógico A0 | Relé → Pino digital 7
// Exemplo: Controle Automatico de Iluminacao com LDR
const int PINO_LDR = A0;
const int PINO_RELE = 7;
const int LIMIAR_ESCURO = 400; // Ajuste conforme o ambiente

void setup() {
  Serial.begin(9600);
  pinMode(PINO_RELE, OUTPUT);
}

void loop() {
  int luminosidade = analogRead(PINO_LDR); // 0 (escuro) a 1023 (claro)

  Serial.print("Luminosidade: ");
  Serial.println(luminosidade);

  if (luminosidade < LIMIAR_ESCURO) {
    digitalWrite(PINO_RELE, HIGH); // Aciona a lampada
  } else {
    digitalWrite(PINO_RELE, LOW);  // Desliga a lampada
  }

  delay(1000);
}

MQ-2 - Detecção de Gás e Fumaça

Monitora a concentração de gás e dispara um alarme sonoro em caso de vazamento.

Ligações: Saída analógica do MQ-2 → Pino A0 | Buzzer → Pino digital 8
// Exemplo: Deteccao de Gas/Fumaca com o sensor MQ-2
const int PINO_MQ2 = A0;
const int LIMIAR_GAS = 300; // Ajustar conforme calibracao do ambiente
const int PINO_BUZZER = 8;

void setup() {
  Serial.begin(9600);
  pinMode(PINO_BUZZER, OUTPUT);
  Serial.println("Aquecendo o sensor MQ-2...");
  delay(20000); // Tempo de pre-aquecimento recomendado
}

void loop() {
  int nivelGas = analogRead(PINO_MQ2); // Leitura analogica (0-1023)

  Serial.print("Nivel de gas: ");
  Serial.println(nivelGas);

  if (nivelGas > LIMIAR_GAS) {
    digitalWrite(PINO_BUZZER, HIGH); // Aciona o alarme sonoro
    Serial.println("ALERTA: Vazamento de gas detectado!");
  } else {
    digitalWrite(PINO_BUZZER, LOW);
  }

  delay(1000);
}

Umidade do Solo - Irrigação Automática

Liga uma bomba de irrigação automaticamente quando o solo está seco.

Ligações: Saída analógica do sensor → Pino A0 | Módulo relé/bomba → Pino digital 6
// Exemplo: Irrigacao Automatica com Sensor Capacitivo de Umidade do Solo
const int PINO_SENSOR = A0;
const int PINO_BOMBA = 6;

// Valores de calibracao (ajustar conforme o sensor utilizado)
const int SOLO_SECO = 800;
const int SOLO_MOLHADO = 300;

void setup() {
  Serial.begin(9600);
  pinMode(PINO_BOMBA, OUTPUT);
}

void loop() {
  int leitura = analogRead(PINO_SENSOR);
  int umidadePercentual = map(leitura, SOLO_SECO, SOLO_MOLHADO, 0, 100);
  umidadePercentual = constrain(umidadePercentual, 0, 100);

  Serial.print("Umidade do solo: ");
  Serial.print(umidadePercentual);
  Serial.println(" %");

  if (umidadePercentual < 30) {
    digitalWrite(PINO_BOMBA, HIGH); // Liga a bomba de irrigacao
  } else {
    digitalWrite(PINO_BOMBA, LOW);  // Desliga a bomba
  }

  delay(2000);
}

YF-S201 - Medição de Vazão de Água

Calcula a vazão de água em litros por minuto contando os pulsos do sensor.

Ligações: Sinal do YF-S201 → Pino digital 2 (interrupção)
// Exemplo: Medicao de Vazao de Agua com o sensor YF-S201
const byte PINO_FLUXO = 2; // Deve ser um pino de interrupcao
volatile int contadorPulsos = 0;
unsigned long ultimoTempo = 0;
float vazaoLmin = 0;

void contarPulso() {
  contadorPulsos++; // Incrementa a cada pulso gerado pelo sensor
}

void setup() {
  Serial.begin(9600);
  pinMode(PINO_FLUXO, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(PINO_FLUXO), contarPulso, RISING);
}

void loop() {
  if (millis() - ultimoTempo >= 1000) { // Calcula a cada 1 segundo
    detachInterrupt(digitalPinToInterrupt(PINO_FLUXO));

    // Formula do fabricante: frequencia (Hz) / 7,5 = vazao em L/min
    vazaoLmin = (contadorPulsos / 7.5);

    Serial.print("Vazao: ");
    Serial.print(vazaoLmin);
    Serial.println(" L/min");

    contadorPulsos = 0;
    ultimoTempo = millis();
    attachInterrupt(digitalPinToInterrupt(PINO_FLUXO), contarPulso, RISING);
  }
}

ACS712 - Medição de Corrente Elétrica

Calcula a corrente elétrica em Amperes a partir da tensão de saída do sensor.

Ligações: Saída OUT do ACS712 → Pino analógico A0
// Exemplo: Medicao de Corrente Eletrica com o sensor ACS712 (modulo 20A)
const int PINO_ACS712 = A0;
const float SENSIBILIDADE = 0.100;   // 100 mV/A para o modulo de 20A
const float TENSAO_REFERENCIA = 2.5; // Tensao de saida para corrente = 0A

void setup() {
  Serial.begin(9600);
}

void loop() {
  int leituraADC = analogRead(PINO_ACS712);
  float tensao = (leituraADC * 5.0) / 1023.0;
  float corrente = (tensao - TENSAO_REFERENCIA) / SENSIBILIDADE;

  Serial.print("Corrente: ");
  Serial.print(corrente);
  Serial.println(" A");

  delay(500);
}

Sensor Hall A3144 - Medição de RPM

Conta os pulsos gerados por um ímã fixado ao eixo para calcular a rotação por minuto.

Ligações: Saída do A3144 → Pino digital 2 (interrupção)
// Exemplo: Medicao de Rotacao (RPM) com Sensor Hall A3144
const byte PINO_HALL = 2; // Pino de interrupcao
volatile int pulsos = 0;
unsigned long ultimaLeitura = 0;

void contarPulsoHall() {
  pulsos++;
}

void setup() {
  Serial.begin(9600);
  pinMode(PINO_HALL, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(PINO_HALL), contarPulsoHall, FALLING);
}

void loop() {
  if (millis() - ultimaLeitura >= 1000) {
    // Considerando 1 ima fixado ao eixo (1 pulso por volta)
    int rpm = pulsos * 60;

    Serial.print("Rotacao: ");
    Serial.print(rpm);
    Serial.println(" RPM");

    pulsos = 0;
    ultimaLeitura = millis();
  }
}

MFRC522 - Controle de Acesso RFID

Lê o identificador (UID) de cartões RFID aproximados do módulo leitor.

Ligações: SDA→10, SCK→13, MOSI→11, MISO→12, RST→9, GND→GND, 3.3V→3.3V
// Exemplo: Controle de Acesso com o leitor RFID MFRC522
#include <SPI.h>
#include <MFRC522.h>

#define PINO_SS   10
#define PINO_RST  9

MFRC522 leitorRFID(PINO_SS, PINO_RST);

void setup() {
  Serial.begin(9600);
  SPI.begin();
  leitorRFID.PCD_Init(); // Inicializa o modulo MFRC522
  Serial.println("Aproxime o cartao do leitor...");
}

void loop() {
  // Verifica se ha um novo cartao presente
  if (!leitorRFID.PICC_IsNewCardPresent() || !leitorRFID.PICC_ReadCardSerial()) {
    return;
  }

  Serial.print("UID do cartao: ");
  for (byte i = 0; i < leitorRFID.uid.size; i++) {
    Serial.print(leitorRFID.uid.uidByte[i], HEX);
    Serial.print(" ");
  }
  Serial.println();

  leitorRFID.PICC_HaltA(); // Encerra a leitura do cartao atual
}

Encoder KY-040 - Leitura de Posição

Identifica o sentido de rotação e conta pulsos para determinar a posição do eixo.

Ligações: CLK→2, DT→3, SW→4, +→5V, GND→GND
// Exemplo: Leitura de Posicao com o Encoder Rotativo KY-040
const int PINO_CLK = 2;
const int PINO_DT = 3;
const int PINO_SW = 4;

int posicao = 0;
int ultimoEstadoCLK;

void setup() {
  Serial.begin(9600);
  pinMode(PINO_CLK, INPUT);
  pinMode(PINO_DT, INPUT);
  pinMode(PINO_SW, INPUT_PULLUP);

  ultimoEstadoCLK = digitalRead(PINO_CLK);
}

void loop() {
  int estadoCLK = digitalRead(PINO_CLK);

  // Detecta uma nova transicao no canal CLK
  if (estadoCLK != ultimoEstadoCLK) {
    // Compara CLK e DT para identificar o sentido de rotacao
    if (digitalRead(PINO_DT) != estadoCLK) {
      posicao++; // Sentido horario
    } else {
      posicao--; // Sentido anti-horario
    }
    Serial.print("Posicao: ");
    Serial.println(posicao);
  }
  ultimoEstadoCLK = estadoCLK;

  if (digitalRead(PINO_SW) == LOW) {
    Serial.println("Botao pressionado - posicao zerada!");
    posicao = 0;
    delay(300); // Debounce simples
  }
}
